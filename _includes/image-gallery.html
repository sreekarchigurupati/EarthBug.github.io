{%- comment -%} Build a sortable list of "::" rows so we can order photos by date (newest first). Photos without a date in _data/gallery.yml get key 00000000 and fall to the end. {%- endcomment -%} {%- assign rows = "" -%} {%- for file in site.static_files -%} {%- if file.path contains include.folder -%} {%- assign ext = file.extname | downcase -%} {%- if ext == '.jpg' or ext == '.jpeg' or ext == '.png' or ext == '.webp' -%} {%- assign basename = file.path | split: "/" | last -%} {%- assign meta = site.data.gallery | where: "file", basename | first -%} {%- if meta.date -%} {%- assign sortkey = meta.date | date: "%Y%m%d" -%} {%- else -%} {%- assign sortkey = "00000000" -%} {%- endif -%} {%- assign rows = rows | append: sortkey | append: "::" | append: file.path | append: "@@" -%} {%- endif -%} {%- endif -%} {%- endfor -%} {%- assign items = rows | split: "@@" | sort | reverse -%} {%- for item in items -%} {%- unless item == "" -%} {%- assign path = item | split: "::" | last -%} {%- assign basename = path | split: "/" | last -%} {%- assign meta = site.data.gallery | where: "file", basename | first -%} {%- if meta.caption -%} {%- assign caption = meta.caption -%} {%- else -%} {%- assign caption = basename | split: "." | first | replace: '-', ' ' | replace: '_', ' ' -%} {%- endif -%}
